168  <script>
169      // ✅ Solo cambia estas URLs cuando quieras cambiar fotos:
170      const gImages = [
171          "https://i.imgur.com/mzp7NXg.jpg",
172          "https://i.imgur.com/n5aqkKn.jpg",
173          "https://i.imgur.com/jmvL6TT.jpg",
174          "https://i.imgur.com/NtaMwrK.jpg",
175          "https://i.imgur.com/YEB3pfo.jpg",
176          "https://i.imgur.com/DctSdGc.jpg",
177          "https://i.imgur.com/iRViDRi.jpg",
178          "https://i.imgur.com/h91sjnw.jpg"
179      ];
180  
181      let gIndex = 0;
182  
183      const mainImg = document.getElementById("gMainImage");
184      const thumbsWrap = document.getElementById("gThumbs");
185  
186      const lb = document.getElementById("gLightbox");
187      const lbImg = document.getElementById("gLbImage");
188      const lbClose = document.getElementById("gLbClose");
189  
190      function renderThumbs() {
191          thumbsWrap.innerHTML = "";
192          gImages.forEach((src, i) => {
193              const t = document.createElement("div");
194              t.className = "g-thumb" + (i === gIndex ? " active" : "");
195              t.innerHTML = `<img src="${src}" alt="Miniatura ${i+1}">`;
196              t.addEventListener("click", () => setImage(i));
197              thumbsWrap.appendChild(t);
198          });
199      }
200  
201      function setImage(i) {
202          gIndex = (i + gImages.length) % gImages.length;
203          mainImg.src = gImages[gIndex];
204          renderThumbs();
205      }
206  
207      function nextImg() { setImage(gIndex + 1); }
208      function prevImg() { setImage(gIndex - 1); }
209  
210      document.querySelector(".g-next").addEventListener("click", nextImg);
211      document.querySelector(".g-prev").addEventListener("click", prevImg);
212  
213      // Lightbox
214      mainImg.addEventListener("click", () => {
215          lbImg.src = gImages[gIndex];
216          lb.classList.add("open");
217          lb.setAttribute("aria-hidden", "false");
218      });
219  
220      lbClose.addEventListener("click", closeLb);
221      lb.addEventListener("click", (e) => { if (e.target === lb) closeLb(); });
222  
223      function closeLb() {
224          lb.classList.remove("open");
225          lb.setAttribute("aria-hidden", "true");
226          lbImg.src = "";
227      }
228  
229      // teclado (opcional pero bonito)
230      document.addEventListener("keydown", (e) => {
231          if (lb.classList.contains("open")) {
232              if (e.key === "Escape") closeLb();
233              if (e.key === "ArrowRight") nextImg();
234              if (e.key === "ArrowLeft") prevImg();
235          }
236      });
237  
238      // inicio
239      setImage(0);
240  </script>
